package compiler

import (
	"testing"

	"github.com/microsoft/typescript-go/internal/module"
)

func TestReuseLoadedImport(t *testing.T) {
	for _, tc := range []struct {
		name   string
		mutate func(*parseTaskData, *parseTask)
		depth  int
		want   bool
	}{
		{"same path and depth", nil, 2, true},
		{"already visited at shallower depth", nil, 3, true},
		{"shallower revisit", nil, 1, false},
		{"pending load", func(d *parseTaskData, _ *parseTask) { d.tasks["a.ts"].loaded = false }, 2, false},
		{"new spelling", func(_ *parseTaskData, t *parseTask) { t.normalizedFilePath = "A.ts" }, 2, false},
		{"other known spelling", func(d *parseTaskData, _ *parseTask) { d.tasks["A.ts"] = &parseTask{} }, 2, false},
		{"new package id", func(_ *parseTaskData, t *parseTask) { t.packageId.Name = "pkg" }, 2, false},
		{"already recorded package id", func(d *parseTaskData, t *parseTask) {
			d.packageId = module.PackageId{Name: "pkg"}
			t.packageId.Name = "other"
		}, 2, true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			first := &parseTask{normalizedFilePath: "a.ts", loaded: true}
			data := &parseTaskData{tasks: map[string]*parseTask{"a.ts": first}, lowestDepth: 2}
			edge := &parseTask{normalizedFilePath: "a.ts"}
			if tc.mutate != nil {
				tc.mutate(data, edge)
			}
			if got := data.reuseLoadedTask(edge, tc.depth); got != tc.want {
				t.Fatalf("reused = %t, want %t", got, tc.want)
			}
			if tc.want && edge.loadedTask != first || !tc.want && edge.loadedTask != nil {
				t.Fatal("unexpected target after reuse attempt")
			}
		})
	}
}

func TestReuseLoadedImportWhileLocked(t *testing.T) {
	data := &parseTaskData{}
	data.mu.Lock()
	defer data.mu.Unlock()
	if data.reuseLoadedTask(&parseTask{}, 0) {
		t.Fatal("a locked path must follow the regular queued path")
	}
}
